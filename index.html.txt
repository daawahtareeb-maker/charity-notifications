  <script>
    function authenticate() {
      var btn = document.getElementById('login-btn');
      var user = document.getElementById('username').value.trim();
      var pass = document.getElementById('password').value.trim();
      
      if(user === "" || pass === "") {
        alert("الرجاء إدخال اسم المستخدم وكلمة المرور");
        return;
      }
      
      btn.querySelector('span').innerText = 'جاري التحقق...';
      btn.disabled = true;
      
      google.script.run
        .withSuccessHandler(function(response) {
          if(response.success) {
            btn.querySelector('span').innerText = 'تم الدخول بنجاح!';
            btn.style.background = '#2e7d32';
            
            setTimeout(function() {
              document.getElementById('login-panel').style.display = 'none';
              var dashboard = document.getElementById('dashboard-panel');
              dashboard.style.display = 'block';
              
              setTimeout(function() {
                dashboard.style.opacity = '1';
              }, 50);
            }, 500);
            
          } else {
            alert(response.message);
            btn.querySelector('span').innerText = 'دخول';
            btn.disabled = false;
          }
        })
        .withFailureHandler(function(error) {
          alert("حدث خطأ في الاتصال بالخادم");
          btn.querySelector('span').innerText = 'دخول';
          btn.disabled = false;
        })
        .checkLogin(user, pass);
    }

    function sendNotification() {
      var topic = document.getElementById('topic-select').value;
      var title = document.getElementById('msg-title').value.trim();
      var body = document.getElementById('msg-body').value.trim();
      
      if(title === "" || body === "") {
        alert("الرجاء كتابة عنوان وتفاصيل الإشعار");
        return;
      }
      
      var btn = document.getElementById('send-btn');
      btn.querySelector('span').innerText = 'جاري البث...';
      btn.disabled = true;
      
      google.script.run
        .withSuccessHandler(function(response) {
          alert(response.message);
          if (response.success) {
            document.getElementById('msg-title').value = '';
            document.getElementById('msg-body').value = '';
          }
          btn.querySelector('span').innerText = 'بث الإشعار';
          btn.disabled = false;
        })
        .withFailureHandler(function(error) {
          alert("حدث خطأ أثناء الإرسال.");
          btn.querySelector('span').innerText = 'بث الإشعار';
          btn.disabled = false;
        })
        .broadcastNotification(topic, title, body);
    }

    function checkEnter(event) {
      if (event.key === "Enter") {
        authenticate();
      }
    }
  </script>
